import java.util.Scanner;

public class RailFenceCipher {

    static String encrypt(String text, int depth) {
        StringBuilder[] rails = new StringBuilder[depth];

        for (int i = 0; i < depth; i++)
            rails[i] = new StringBuilder();

        int row = 0, direction = 1;

        for (char c : text.toCharArray()) {
            rails[row].append(c);

            if (row == 0)
                direction = 1;
            else if (row == depth - 1)
                direction = -1;

            row += direction;
        }

        String result = "";

        for (StringBuilder rail : rails)
            result += rail;

        return result;
    }

    static String decrypt(String cipher, int depth) {
        int n = cipher.length();
        char[][] rail = new char[depth][n];
        boolean[][] mark = new boolean[depth][n];

        int row = 0, direction = 1;

        for (int col = 0; col < n; col++) {
            mark[row][col] = true;

            if (row == 0)
                direction = 1;
            else if (row == depth - 1)
                direction = -1;

            row += direction;
        }

        int index = 0;

        for (int i = 0; i < depth; i++) {
            for (int j = 0; j < n; j++) {
                if (mark[i][j])
                    rail[i][j] = cipher.charAt(index++);
            }
        }

        String result = "";
        row = 0;
        direction = 1;

        for (int col = 0; col < n; col++) {
            result += rail[row][col];

            if (row == 0)
                direction = 1;
            else if (row == depth - 1)
                direction = -1;

            row += direction;
        }

        return result;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter message: ");
        String message = sc.nextLine();

        System.out.print("Enter depth: ");
        int depth = sc.nextInt();

        if (depth < 2 || depth > message.length()) {
            System.out.println("Invalid depth.");
        } else {
            String encrypted = encrypt(message, depth);
            String decrypted = decrypt(encrypted, depth);

            System.out.println("Encrypted Message: " + encrypted);
            System.out.println("Decrypted Message: " + decrypted);
        }

        sc.close();
    }
}

import java.util.Scanner;

public class TransCipher {

    static final int COLS = 4;

    static String encrypt(String text) {
        text = text.replaceAll("\\s", "");

        int rows = (int) Math.ceil((double) text.length() / COLS);

        char[][] matrix = new char[rows][COLS];

        int k = 0;

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < COLS; j++) {
                matrix[i][j] = k < text.length()
                        ? text.charAt(k++) : 'X';
            }
        }

        String result = "";

        for (int j = 0; j < COLS; j++) {
            for (int i = 0; i < rows; i++) {
                result += matrix[i][j];
            }
        }

        return result;
    }

    static String decrypt(String cipher, int originalLength) {
        int rows = cipher.length() / COLS;
        char[][] matrix = new char[rows][COLS];

        int k = 0;

        for (int j = 0; j < COLS; j++) {
            for (int i = 0; i < rows; i++) {
                matrix[i][j] = cipher.charAt(k++);
            }
        }

        String result = "";

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < COLS; j++) {
                result += matrix[i][j];
            }
        }

        return result.substring(0, originalLength);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter plain text: ");
        String message = sc.nextLine().replaceAll("\\s", "");

        String encrypted = encrypt(message);
        String decrypted = decrypt(encrypted, message.length());

        System.out.println("Encrypted Message: " + encrypted);
        System.out.println("Decrypted Message: " + decrypted);

        sc.close();
    }
}
