import java.math.BigInteger;
import java.util.Scanner;

public class RSA {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter first prime number: ");
        BigInteger p = sc.nextBigInteger();

        System.out.print("Enter second prime number: ");
        BigInteger q = sc.nextBigInteger();

        System.out.print("Enter message (number): ");
        BigInteger message = sc.nextBigInteger();

        if (!p.isProbablePrime(20) || !q.isProbablePrime(20)
                || p.equals(q)) {
            System.out.println("Enter two different prime numbers.");
            sc.close();
            return;
        }

        BigInteger n = p.multiply(q);

        BigInteger phi = p.subtract(BigInteger.ONE)
                .multiply(q.subtract(BigInteger.ONE));

        BigInteger e = BigInteger.valueOf(2);

        while (!e.gcd(phi).equals(BigInteger.ONE)) {
            e = e.add(BigInteger.ONE);
        }

        BigInteger d = e.modInverse(phi);

        if (message.compareTo(n) >= 0) {
            System.out.println("Message must be smaller than n.");
            sc.close();
            return;
        }

        BigInteger encrypted = message.modPow(e, n);
        BigInteger decrypted = encrypted.modPow(d, n);

        System.out.println("Public Key: (" + e + ", " + n + ")");
        System.out.println("Private Key: (" + d + ", " + n + ")");
        System.out.println("Encrypted Message: " + encrypted);
        System.out.println("Decrypted Message: " + decrypted);

        sc.close();
    }
}
