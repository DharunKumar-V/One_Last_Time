import java.util.Base64;
import java.util.Scanner;
import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;

public class AES {

    static SecretKey key;

    static String encrypt(String message) throws Exception {
        byte[] iv = new byte[12];
        new java.security.SecureRandom().nextBytes(iv);

        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");

        cipher.init(Cipher.ENCRYPT_MODE, key,
                new GCMParameterSpec(128, iv));

        byte[] encrypted = cipher.doFinal(message.getBytes("UTF-8"));

        byte[] result = new byte[iv.length + encrypted.length];

        System.arraycopy(iv, 0, result, 0, iv.length);
        System.arraycopy(encrypted, 0, result, iv.length,
                encrypted.length);

        return Base64.getEncoder().encodeToString(result);
    }

    static String decrypt(String encryptedText) throws Exception {
        byte[] data = Base64.getDecoder().decode(encryptedText);

        byte[] iv = new byte[12];
        byte[] encrypted = new byte[data.length - 12];

        System.arraycopy(data, 0, iv, 0, 12);
        System.arraycopy(data, 12, encrypted, 0, encrypted.length);

        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");

        cipher.init(Cipher.DECRYPT_MODE, key,
                new GCMParameterSpec(128, iv));

        return new String(cipher.doFinal(encrypted), "UTF-8");
    }

    public static void main(String[] args) throws Exception {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter URL or message: ");
        String message = sc.nextLine();

        KeyGenerator generator = KeyGenerator.getInstance("AES");
        generator.init(128);
        key = generator.generateKey();

        String encrypted = encrypt(message);
        String decrypted = decrypt(encrypted);

        System.out.println("Original Message: " + message);
        System.out.println("Encrypted Message: " + encrypted);
        System.out.println("Decrypted Message: " + decrypted);

        sc.close();
    }
}
